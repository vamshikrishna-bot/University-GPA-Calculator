""" University GPA Calculation System Handles student enrollment, grade assignment, and GPA calculation """ class Course: """Represents a university course""" def __init__(self, course_id, name, credits): """ Initialize a course Args: course_id (str): Unique course identifier name (str): Course name credits (int): Number of credit hours """ if credits <= 0: raise ValueError("Credits must be positive") self.course_id = course_id self.name = name self.credits = credits def __repr__(self): return f"Course({self.course_id}, {self.name}, {self.credits} credits)" class Grade: """Represents a grade with validation""" # Grade to GPA mapping (4.0 scale) GRADE_POINTS = { 'A+': 4.0, 'A': 4.0, 'A-': 3.7, 'B+': 3.3, 'B': 3.0, 'B-': 2.7, 'C+': 2.3, 'C': 2.0, 'C-': 1.7, 'D+': 1.3, 'D': 1.0, 'D-': 0.7, 'F': 0.0 } def __init__(self, letter_grade): """ Initialize a grade Args: letter_grade (str): Letter grade (A+, A, A-, B+, etc.) Raises: ValueError: If grade is invalid """ if letter_grade not in self.GRADE_POINTS: raise ValueError(f"Invalid grade: {letter_grade}. " f"Valid grades: {list(self.GRADE_POINTS.keys())}") self.letter_grade = letter_grade self.grade_point = self.GRADE_POINTS[letter_grade] def __repr__(self): return f"Grade({self.letter_grade}, {self.grade_point})" class Student: """Represents a student with course enrollments and grades""" def __init__(self, student_id, name): """ Initialize a student Args: student_id (str): Unique student identifier name (str): Student name """ self.student_id = student_id self.name = name self.enrollments = {} # {course_id: {'course': Course, 'grade': Grade}} def enroll_course(self, course): """ Enroll student in a course Args: course (Course): Course to enroll in """ if course.course_id in self.enrollments: raise ValueError(f"Already enrolled in {course.course_id}") self.enrollments[course.course_id] = { 'course': course, 'grade': None } def assign_grade(self, course_id, letter_grade): """ Assign a grade to a course Args: course_id (str): Course identifier letter_grade (str): Letter grade to assign Raises: ValueError: If not enrolled in course or invalid grade """ if course_id not in self.enrollments: raise ValueError(f"Not enrolled in course {course_id}") grade = Grade(letter_grade) # This validates the grade self.enrollments[course_id]['grade'] = grade def calculate_gpa(self): """ Calculate GPA based on all graded courses Returns: float: GPA on 4.0 scale, or 0.0 if no courses """ total_points = 0.0 total_credits = 0 for enrollment in self.enrollments.values(): grade = enrollment['grade'] course = enrollment['course'] # Only include courses with assigned grades if grade is not None: total_points += grade.grade_point * course.credits total_credits += course.credits # Return 0.0 if no graded courses if total_credits == 0: return 0.0 return round(total_points / total_credits, 2) def generate_transcript(self): """ Generate a formatted transcript Returns: str: Formatted transcript string """ transcript = f"\n{'='*60}\n" transcript += f"TRANSCRIPT - {self.name} (ID: {self.student_id})\n" transcript += f"{'='*60}\n\n" if not self.enrollments: transcript += "No courses enrolled.\n" transcript += f"\nCumulative GPA: 0.00\n" transcript += f"{'='*60}\n" return transcript transcript += f"{'Course ID':<12} {'Course Name':<25} {'Credits':<8} {'Grade':<8}\n" transcript += f"{'-'*60}\n" for course_id, enrollment in self.enrollments.items(): course = enrollment['course'] grade = enrollment['grade'] grade_str = grade.letter_grade if grade else "N/A" transcript += f"{course.course_id:<12} {course.name:<25} " transcript += f"{course.credits:<8} {grade_str:<8}\n" transcript += f"\n{'='*60}\n" transcript += f"Cumulative GPA: {self.calculate_gpa():.2f}\n" transcript += f"{'='*60}\n" return transcript def __repr__(self): return f"Student({self.student_id}, {self.name}, GPA: {self.calculate_gpa()})" # ============================================================================ # UNIT TESTS #
